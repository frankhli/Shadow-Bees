7. React 组件代码

ForeignGuestBadge.tsx

import { ShieldCheck, BadgeCheck, Globe } from 'lucide-react';
import { cn } from '@/lib/utils';

interface ForeignGuestBadgeProps {
  certification: {
    acceptsForeignGuests: boolean;
    certificationType: 'official' | 'verified' | 'self_reported';
    policeRegistration: boolean;
  };
  size?: 'sm' | 'md' | 'lg';
  onClick?: () => void;
  className?: string;
}

const badgeConfig = {
  official: {
    icon: ShieldCheck,
    bg: 'bg-blue-500',
    hoverBg: 'hover:bg-blue-600',
    text: 'Verified Foreign Guest',
  },
  verified: {
    icon: BadgeCheck,
    bg: 'bg-emerald-500',
    hoverBg: 'hover:bg-emerald-600',
    text: 'Accepts Foreigners',
  },
  self_reported: {
    icon: Globe,
    bg: 'bg-gray-500',
    hoverBg: 'hover:bg-gray-600',
    text: 'Foreign Guest OK',
  },
};

const sizeConfig = {
  sm: { padding: 'px-2 py-1', text: 'text-xs', icon: 'w-3 h-3' },
  md: { padding: 'px-2.5 py-1', text: 'text-xs', icon: 'w-3.5 h-3.5' },
  lg: { padding: 'px-3 py-1.5', text: 'text-sm', icon: 'w-4 h-4' },
};

export function ForeignGuestBadge({ 
  certification, 
  size = 'md', 
  onClick,
  className 
}: ForeignGuestBadgeProps) {
  if (!certification.acceptsForeignGuests) return null;
  
  const config = badgeConfig[certification.certificationType];
  const sizeStyles = sizeConfig[size];
  const Icon = config.icon;
  
  return (
    <button
      onClick={onClick}
      className={cn(
        'inline-flex items-center gap-1.5 font-semibold rounded-full',
        'transition-colors cursor-pointer',
        'shadow-[0_2px_8px_rgba(0,0,0,0.15),0_0_0_1px_rgba(255,255,255,0.1)]',
        config.bg,
        config.hoverBg,
        sizeStyles.padding,
        sizeStyles.text,
        className
      )}
    >
      <Icon className={sizeStyles.icon} />
      <span className="text-white">{config.text}</span>
      {certification.policeRegistration && size !== 'sm' && (
        <span className="text-white/70 text-[10px] ml-0.5">(Police Reg.)</span>
      )}
    </button>
  );
}

8. 使用示例

酒店卡片使用

// HotelCard.tsx
import { ForeignGuestBadge } from './ForeignGuestBadge';
import { CertificationModal } from './CertificationModal';
import { useState } from 'react';

export function HotelCard({ hotel }) {
  const [showCertModal, setShowCertModal] = useState(false);

  return (
    <>
      <div className="relative aspect-square rounded-xl overflow-hidden">
        {/* Foreign Guest Badge - Top Left */}
        <div className="absolute top-3 left-3">
          <ForeignGuestBadge
            certification={hotel.certification}
            size="sm"
            onClick={() => setShowCertModal(true)}
          />
        </div>
        
        {/* ... rest of card */}
      </div>

      <CertificationModal
        isOpen={showCertModal}
        onClose={() => setShowCertModal(false)}
        hotelName={hotel.name}
      />
    </>
  );
}
